Function

household <- sort(unique(df_index$household))
household_colors <- hues::iwanthue(length(household))

# Household scale
household_scale <- scale_fill_manual(breaks = household,values = household_colors,guide = guide_legend(ncol=7),name = '1. Index participant') 

# Visit scale
visit_scale <- scale_fill_manual(values = c("black","red","blue","purple"),breaks = c(0,1,2,3),labels = c("Clinical","Baseline","One month","Two month"),name = '2. Visit',guide = guide_legend(nrow=5))

# Prolonged scale
prolonged_colonization_scale <- scale_fill_manual(breaks = c("Yes","No","No index specimen"), values = c("darkred","white","gray"),name = '3. Persistent colonization',guide = guide_legend(nrow=5))

# Function
get_figure <- function(tree, species_df, offset_x, title, offset_y,
                       ylim_decision = NULL, title_margin = -2){
  tree_obj <- ggtree(tree)
  max_y <- max(tree_obj$data$y, na.rm = TRUE)
  if(is.null(ylim_decision)) ylim_decision <- max_y + 1
  
  rownames(species_df) <- species_df$genomic_id
  p0 <- gheatmap(tree_obj, species_df %>% select(household), width=0.1, colnames=F) + household_scale + ggnewscale::new_scale_fill()  
  p1 <- gheatmap(p0, species_df %>% select(visit) %>% mutate_all(as.factor), width=0.1, offset=offset_x, colnames=F) + visit_scale + ggnewscale::new_scale_fill()  
  p2 <- gheatmap(p1, species_df %>% select(prolonged_colonization) %>% mutate_all(as.factor), width=0.1, offset=offset_x*2, colnames=F) + prolonged_colonization_scale  
  p2 + ggtitle(title) + 
    theme(plot.title = element_markdown()) + 
    geom_treescale(x=0, y=0, offset=offset_y) + 
    theme(legend.position='none') + 
    ylim(NA, ylim_decision) + 
    theme(plot.title = element_markdown(margin = margin(b = title_margin, unit = "pt"))) 
}
